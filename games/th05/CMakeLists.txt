cmake_minimum_required(VERSION 3.25)
project(th05_port VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)

if(NOT TARGET pc98_numeric)
	add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../shared"
		"${CMAKE_CURRENT_BINARY_DIR}/pc98-shared")
endif()

add_library(th05_core STATIC src/core/player.cpp src/core/item.cpp src/core/score.cpp
	src/core/shot_patterns.cpp src/core/bomb.cpp src/core/std_load.cpp
	src/core/std_run.cpp src/core/scroll.cpp src/core/stage_init.cpp
	src/core/stage.cpp src/core/enemy.cpp src/core/bullet.cpp src/core/laser.cpp
	src/core/boss.cpp src/core/stage_bonus.cpp src/core/stage_scenery.cpp
	src/core/stage4_projectiles.cpp src/core/sword.cpp src/core/b6ball.cpp src/core/firewave.cpp
	src/core/shinki_background.cpp src/core/midboss_extra.cpp)
target_include_directories(th05_core PUBLIC include)
target_link_libraries(th05_core PUBLIC pc98_numeric)
if(MSVC)
	target_compile_options(th05_core PRIVATE /W4 /WX /permissive- /Zc:preprocessor)
else()
	target_compile_options(th05_core PRIVATE
		-Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
endif()

add_executable(th05-headless src/headless.cpp)
target_link_libraries(th05-headless PRIVATE th05_core)
if(MSVC)
