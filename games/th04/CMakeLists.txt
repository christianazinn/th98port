function(th04_set_warnings target)
	if(MSVC)
		target_compile_options(${target} PRIVATE /W4 /WX /permissive- /Zc:preprocessor)
	else()
		target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
	endif()
endfunction()
th04_set_warnings(th04_core)

target_link_libraries(th04-headless PRIVATE th04_core)
